cmake_minimum_required(VERSION 3.24)
project(DcsEsportServer VERSION 0.4.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BOOST_ROOT "${CMAKE_SOURCE_DIR}/.tools/boost_1_90_0" CACHE PATH "Pinned Boost headers")
find_path(BOOST_INCLUDE_DIR boost/json/src.hpp HINTS "${BOOST_ROOT}" REQUIRED)
set(OPENSSL_USE_STATIC_LIBS TRUE)
find_package(OpenSSL 3.0 REQUIRED)
add_executable(dcs-esport-server src/server/main.cpp src/server/boost_json.cpp)
target_include_directories(dcs-esport-server SYSTEM PRIVATE "${BOOST_INCLUDE_DIR}")
target_compile_definitions(dcs-esport-server PRIVATE BOOST_JSON_NO_LIB BOOST_ALL_NO_LIB)
target_link_libraries(dcs-esport-server PRIVATE OpenSSL::SSL OpenSSL::Crypto)
if(WIN32)
  target_compile_definitions(dcs-esport-server PRIVATE _WIN32_WINNT=0x0601 NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(dcs-esport-server PRIVATE ws2_32 mswsock crypt32 bcrypt)
endif()
if(MSVC)
  target_compile_options(dcs-esport-server PRIVATE /W4 /utf-8 /EHsc)
else()
  target_compile_options(dcs-esport-server PRIVATE -Wall -Wextra -Wpedantic)
  if(WIN32 AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    target_link_options(dcs-esport-server PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
endif()
install(TARGETS dcs-esport-server RUNTIME DESTINATION .)
install(FILES "${BOOST_INCLUDE_DIR}/LICENSE_1_0.txt" DESTINATION .)
install(FILES "${CMAKE_SOURCE_DIR}/docs/OpenSSL-LICENSE.txt" DESTINATION .)
